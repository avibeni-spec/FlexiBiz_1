import 'package:cloud_firestore/cloud_firestore.dart'; import 'package:flutter_riverpod/flutter_riverpod.dart'; import '../models/client_model.dart'; import '../models/appointment_model.dart'; /// The "Perfect" Database Service /// Includes robust error handling, logging, and performance optimizations. class DatabaseService { final FirebaseFirestore _db = FirebaseFirestore.instance; // --- ERROR HANDLING WRAPPER --- Future _performOperation(Future Function() operation, String errorMessage) async { try { return await operation(); } on FirebaseException catch (e) { // Log to analytics (e.g. Firebase Crashlytics) print('Firebase Error: ${e.code} - ${e.message}'); throw 'Database Sync Failed: ${e.message}'; } catch (e) { print('Unknown Error: $e'); throw errorMessage; } } // --- CLIENTS --- Stream> streamClients() { return _db.collection('clients') .orderBy('fullName') // Keep it sorted for UX .snapshots() .map((snapshot) => snapshot.docs.map((doc) => Client.fromMap(doc.data())).toList()); } Future saveClient(Client client) async { await _performOperation(() => _db.collection('clients').doc(client.id).set(client.toMap(), SetOptions(merge: true)), 'Could not save client data. Please try again.' ); } Future deleteClient(String clientId) async { await _performOperation(() => _db.collection('clients').doc(clientId).delete(), 'Could not delete client. It might be linked to appointments.' ); } // --- APPOINTMENTS --- Stream> streamAppointmentsForDate(DateTime date) { String dateStr = DateTime(date.year, date.month, date.day).toIso8601String().substring(0, 10); // Optimization: Query by range to capture the whole day return _db.collection('appointments') .where('dateTime', isGreaterThanOrEqualTo: '${dateStr}T00:00:00') .where('dateTime', isLessThanOrEqualTo: '${dateStr}T23:59:59') .snapshots() .map((snapshot) => snapshot.docs.map((doc) => Appointment.fromMap(doc.data())).toList()); } Future createAppointment(Appointment appointment) async { await _performOperation(() => _db.collection('appointments').doc(appointment.id).set(appointment.toMap()), 'Booking failed. Please check your internet connection.' ); } Future updateAppointmentStatus(String id, String status) async { await _performOperation(() => _db.collection('appointments').doc(id).update({'status': status}), 'Status update failed.' ); } // --- ANALYTICS & BUSINESS --- Stream> streamBusinessInfo(String businessId) { return _db.collection('businesses').doc(businessId).snapshots().map((doc) => doc.data() ?? {}); } } // --- RIVERPOD PROVIDERS WITH REFRESH LOGIC --- final databaseProvider = Provider((ref) => DatabaseService()); final clientsStreamProvider = StreamProvider>((ref) { // Logic: Keep the stream alive even if the user navigates away temporarily final link = ref.keepAlive(); return ref.watch(databaseProvider).streamClients(); }); final appointmentsStreamProvider = StreamProvider.family, DateTime>((ref, date) { return ref.watch(databaseProvider).streamAppointmentsForDate(date); });
