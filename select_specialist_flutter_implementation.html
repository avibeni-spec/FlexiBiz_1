import 'package:flutter/material.dart'; import 'package:flutter_riverpod/flutter_riverpod.dart'; import '../../data/models/specialist_model.dart'; // Specialist model defined in specs import '../../logic/providers/database_providers.dart'; class ClientSelectExpertScreen extends ConsumerStatefulWidget { const ClientSelectExpertScreen({super.key}); @override ConsumerState createState() => _ClientSelectExpertScreenState(); } class _ClientSelectExpertScreenState extends ConsumerState { String _selectedCategory = 'All'; @override Widget build(BuildContext context) { // Watch the real-time stream of experts final specialistsAsync = ref.watch(specialistsStreamProvider); return Scaffold( backgroundColor: const Color(0xFF131313), appBar: _buildAppBar(), body: specialistsAsync.when( data: (specialists) { // Local filtering logic final filtered = _selectedCategory == 'All' ? specialists : specialists.where((s) => s.expertise.contains(_selectedCategory)).toList(); return CustomScrollView( slivers: [ SliverPadding( padding: const EdgeInsets.symmetric(horizontal: 24, vertical: 32), sliver: SliverToBoxAdapter( child: Column( crossAxisAlignment: CrossAxisAlignment.start, children: [ _buildHeader(), const SizedBox(height: 32), _buildCategoryFilters(), ], ), ), ), SliverPadding( padding: const EdgeInsets.symmetric(horizontal: 24), sliver: SliverList( delegate: SliverChildBuilderDelegate( (context, index) => _buildExpertCard(filtered[index]), childCount: filtered.length, ), ), ), const SliverToBoxAdapter(child: SizedBox(height: 100)), ], ); }, loading: () => const Center(child: CircularProgressIndicator(color: Color(0xFFE9C349))), error: (err, stack) => Center(child: Text('Error loading experts: $err', style: const TextStyle(color: Colors.red))), ), bottomNavigationBar: _buildBottomNavBar(), ); } PreferredSizeWidget _buildAppBar() { return AppBar( backgroundColor: Colors.transparent, elevation: 0, leading: IconButton(icon: const Icon(Icons.arrow_back, color: Colors.white), onPressed: () => Navigator.pop(context)), title: const Text( 'SELECT EXPERT', style: TextStyle(fontFamily: 'Manrope', fontWeight: FontWeight.w300, letterSpacing: 4.0, fontSize: 16, color: Colors.white), ), centerTitle: true, ); } Widget _buildHeader() { return Column( crossAxisAlignment: CrossAxisAlignment.start, children: [ const Text( 'Atelier Collective', style: TextStyle(color: Color(0xFFE9C349), fontSize: 12, letterSpacing: 2, fontWeight: FontWeight.bold), ), const SizedBox(height: 12), const Text( 'Curated Talent\nFor Your Needs', style: TextStyle(color: Colors.white, fontSize: 36, fontWeight: FontWeight.w300, height: 1.1), ), ], ); } Widget _buildCategoryFilters() { final categories = ['All', 'Skin', 'Hair', 'Massage', 'Wellness']; return SingleChildScrollView( scrollDirection: Axis.horizontal, child: Row( children: categories.map((cat) => _buildFilterChip(cat)).toList(), ), ); } Widget _buildFilterChip(String label) { final isSelected = _selectedCategory == label; return GestureDetector( onTap: () => setState(() => _selectedCategory = label), child: Container( margin: const EdgeInsets.only(right: 12), padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 12), decoration: BoxDecoration( color: isSelected ? Colors.white : const Color(0xFF1C1B1B), borderRadius: BorderRadius.circular(40), border: Border.all(color: isSelected ? Colors.white : Colors.white.withOpacity(0.1)), ), child: Text( label, style: TextStyle( color: isSelected ? Colors.black : Colors.white.withOpacity(0.4), fontWeight: isSelected ? FontWeight.bold : FontWeight.normal, fontSize: 12, ), ), ), ); } Widget _buildExpertCard(Specialist s) { return Container( margin: const EdgeInsets.only(bottom: 24), padding: const EdgeInsets.all(24), decoration: BoxDecoration( color: const Color(0xFF1C1B1B), borderRadius: BorderRadius.circular(32), ), child: Row( children: [ _buildSpecialistImage(s.imageUrl), const SizedBox(width: 20), Expanded( child: Column( crossAxisAlignment: CrossAxisAlignment.start, children: [ Text(s.name, style: const TextStyle(color: Colors.white, fontSize: 20, fontWeight: FontWeight.w400)), const SizedBox(height: 4), Text(s.role, style: TextStyle(color: Colors.white.withOpacity(0.4), fontSize: 12)), const SizedBox(height: 12), Row( children: [ const Icon(Icons.star, color: Color(0xFFE9C349), size: 14), const SizedBox(width: 4), Text(s.rating.toString(), style: const TextStyle(color: Colors.white, fontSize: 12, fontWeight: FontWeight.bold)), ], ), ], ), ), _buildSelectButton(s), ], ), ); } Widget _buildSpecialistImage(String url) { return Container( padding: const EdgeInsets.all(2), decoration: BoxDecoration(shape: BoxShape.circle, border: Border.all(color: const Color(0xFFE9C349), width: 1)), child: CircleAvatar( radius: 35, backgroundImage: NetworkImage(url), ), ); } Widget _buildSelectButton(Specialist s) { return InkWell( onTap: () { // Logic: Pass s.id to the booking state Navigator.pop(context, s.id); }, child: Container( padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8), decoration: BoxDecoration(border: Border.all(color: Colors.white.withOpacity(0.2)), borderRadius: BorderRadius.circular(12)), child: const Text('SELECT', style: TextStyle(color: Colors.white, fontSize: 10, letterSpacing: 1, fontWeight: FontWeight.bold)), ), ); } Widget _buildBottomNavBar() { return Container( padding: const EdgeInsets.only(bottom: 24, left: 24, right: 24, top: 12), color: const Color(0xFF131313), child: Container( height: 72, decoration: BoxDecoration(color: const Color(0xFF1C1B1B), borderRadius: BorderRadius.circular(40)), child: Row( mainAxisAlignment: MainAxisAlignment.spaceAround, children: [ _buildNavItem(Icons.grid_view, false), _buildNavItem(Icons.face_retouching_natural, true), _buildNavItem(Icons.event_note, false), _buildNavItem(Icons.person_outline, false), ], ), ), ); } Widget _buildNavItem(IconData icon, bool isActive) { return Icon(icon, color: isActive ? const Color(0xFFE9C349) : Colors.white.withOpacity(0.3), size: 24); } }
