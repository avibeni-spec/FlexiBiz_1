# FlexiBiz - PDF Report Generation Technical Specification This document outlines how to implement the "Download PDF Report" feature in the Aura Analytics Dashboard using the `pdf` and `printing` packages. ## 1. Dependencies Add the following to your `pubspec.yaml`: ```yaml dependencies: pdf: ^3.10.0 printing: ^5.11.0 path_provider: ^2.1.0 ``` ## 2. PDF Service Implementation (`pdf_service.dart`) This service handles the creation of the PDF document, including drawing the charts and tables. ```dart import 'dart:io'; import 'package:pdf/pdf.dart'; import 'package:pdf/widgets.dart' as pw; import 'package:printing/printing.dart'; import 'package:path_provider/path_provider.dart'; class PdfReportService { static Future generateAndDownloadReport(Map analyticsData) async { final pdf = pw.Document(); // Load custom fonts for brand consistency final font = await PdfGoogleFonts.manropeLight(); final fontBold = await PdfGoogleFonts.manropeBold(); pdf.addPage( pw.MultiPage( pageFormat: PdfPageFormat.a4, theme: pw.ThemeData.withFont(base: font, bold: fontBold), build: (context) => [ _buildHeader(analyticsData['businessName']), pw.SizedBox(height: 40), _buildPerformanceSummary(analyticsData), pw.SizedBox(height: 40), _buildRevenueTable(analyticsData['monthlyRevenue']), pw.SizedBox(height: 40), _buildFooter(), ], ), ); // Output the PDF await Printing.layoutPdf(onLayout: (PdfPageFormat format) async => pdf.save()); // Alternatively, save to file // final output = await getTemporaryDirectory(); // final file = File("${output.path}/FlexiBiz_Report.pdf"); // await file.writeAsBytes(await pdf.save()); } static pw.Widget _buildHeader(String businessName) { return pw.Row( mainAxisAlignment: pw.MainAxisAlignment.spaceBetween, children: [ pw.Column( crossAxisAlignment: pw.CrossAxisAlignment.start, children: [ pw.Text('FLEXIBIZ EXECUTIVE REPORT', style: pw.TextStyle(color: PdfColors.grey700, fontSize: 10, letterSpacing: 2)), pw.SizedBox(height: 8), pw.Text(businessName.toUpperCase(), style: pw.TextStyle(fontSize: 24, fontWeight: pw.FontWeight.bold)), ], ), pw.Text(DateTime.now().toString().substring(0, 10), style: pw.TextStyle(color: PdfColors.grey)), ], ); } static pw.Widget _buildPerformanceSummary(Map data) { return pw.Column( crossAxisAlignment: pw.CrossAxisAlignment.start, children: [ pw.Text('PERFORMANCE SUMMARY', style: pw.TextStyle(fontWeight: pw.FontWeight.bold, fontSize: 14)), pw.SizedBox(height: 20), pw.Row( mainAxisAlignment: pw.MainAxisAlignment.spaceAround, children: [ _buildStatItem('Total Revenue', data['totalRevenue']), _buildStatItem('New Clients', data['newClients'].toString()), _buildStatItem('Retention', '${data['retentionRate']}%'), ], ), ], ); } static pw.Widget _buildStatItem(String label, String value) { return pw.Column( children: [ pw.Text(label, style: pw.TextStyle(color: PdfColors.grey, fontSize: 10)), pw.SizedBox(height: 8), pw.Text(value, style: pw.TextStyle(fontSize: 18, fontWeight: pw.FontWeight.bold, color: PdfColors.black)), ], ); } static pw.Widget _buildRevenueTable(List monthlyData) { return pw.TableHelper.fromTextArray( context: null, data: >[ ['Month', 'Gross Revenue', 'Appointments', 'Growth'], ...monthlyData.map((item) => [item['month'], item['revenue'], item['count'], item['growth']]) ], headerStyle: pw.TextStyle(color: PdfColors.white, fontWeight: pw.FontWeight.bold), headerDecoration: pw.BoxDecoration(color: PdfColor.fromHex('#131313')), cellAlignment: pw.Alignment.centerLeft, cellStyle: const pw.TextStyle(fontSize: 10), ); } static pw.Widget _buildFooter() { return pw.Container( alignment: pw.Alignment.centerRight, margin: const pw.EdgeInsets.only(top: 1.0 * PdfPageFormat.cm), child: pw.Text('Generated by Aura Analytics Suite', style: pw.TextStyle(color: PdfColors.grey, fontSize: 8)), ); } } ``` ## 3. Integrating with the Dashboard UI In your `DashboardScreen`, trigger the service when the button is pressed: ```dart // Inside the TopAppBar implementation IconButton( icon: const Icon(Icons.picture_as_pdf_outlined, color: Color(0xFFE9C349)), onPressed: () async { // 1. Show Global Loading ref.read(globalLoadingProvider.notifier).state = true; try { // 2. Fetch data (or use existing state) final reportData = { 'businessName': 'Onyx Studio', 'totalRevenue': '\$142,850', 'newClients': 184, 'retentionRate': 68, 'monthlyRevenue': [ {'month': 'Jan', 'revenue': '\$22,400', 'count': '112', 'growth': '+4%'}, // ... more data ], }; // 3. Generate PDF await PdfReportService.generateAndDownloadReport(reportData); } catch (e) { debugPrint('Error generating PDF: $e'); } finally { // 4. Hide Global Loading ref.read(globalLoadingProvider.notifier).state = false; } }, ), ```
